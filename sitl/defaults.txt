# Minimum BF SITL config for Project_Beta_Ardu sim testing.
# Each setting here MUST be valid in BF 4.5.1 — the build pin in the
# Dockerfile. If you bump BETAFLIGHT_REF, re-run start.sh and audit the
# stderr for INVALID NAME / INVALID VALUE messages, then update.
#
# Why so short: previous versions of this file shipped a comprehensive
# Phase 0 GPS Rescue config block. BF 4.5.1 renamed half those settings
# and removed the others, AND something in the larger block segfaulted
# the BF SITL binary during `save`. Build back up to a richer config one
# verified-working line at a time.

# 255 = 0xFF = channels 1-8 all MSP-overrideable. The companion only
# touches 1-4, but hover_test / mission_demo / radio_to_bf need AUX1
# (channel 5) to drive the ARM box switch. mask=15 only covered 1-4
# and silently dropped our arm signal.
set msp_override_channels_mask = 255
set mag_hardware = NONE
# Without a motor protocol BF refuses to arm with MOTOR_PROTO flag.
# PWM works in SITL where DSHOT/proshot don't (no real motor h/w).
set motor_pwm_protocol = PWM
# Disable real-flight pre-arm checks that don't apply in sim:
# - small_angle=180 lets us arm at any orientation (no IMU needed up).
# - runaway_takeoff_prevention=OFF skips the autodetect that cuts motors
#   on uncommanded takeoff (BF defaults are tuned for real hardware).
# DO NOT raise min_check/max_check here. Default min_check=1050 is the
# threshold for "throttle is in idle band" — when throttle < min_check
# BF clears the THROTTLE arming flag. Our hover_test sends throttle=1000
# at idle, which IS < 1050 → flag clears as intended. Setting
# min_check=1000 (which I tried) makes the check `1000 < 1000` which
# fails → THROTTLE flag never clears → ARM_SWITCH fires when AUX1 goes
# high → motors stay 0. Verified live via integrations/tools/bf_diag.
set small_angle = 180
set runaway_takeoff_prevention = OFF
# Bind ARM (mode index 0) to AUX1 (channel 5 = aux idx 0) high band
# 1700-2100. Without this, even with mask=255 BF has nothing watching
# AUX1 to flip the ARM box, so AUX1=2000 from hover_test is ignored
# and we never arm. `aux 0 0 0 1700 2100 0 0` = "ARM mode active when
# AUX1 is in [1700,2100]".
aux 0 0 0 1700 2100 0 0
